    ```python
    @app.route("/automation/delays", methods=["POST"])
    def automation_delays():
        payload = request.get_json(silent=True) or request.form.to_dict()
        try:
            form = AutomationDelaysForm(**payload)
        except Exception as exc:
            return jsonify({"success": False, "error": str(exc)}), 200
        runtime_state = load_runtime_state()
        runtime_state.return_to_live_delay = form.return_to_live_delay
        runtime_state.gameplay_hold_delay = form.gameplay_hold_delay
        save_runtime_state(runtime_state)
        socketio.emit("automation_delays_updated", {
            "return_to_live_delay": form.return_to_live_delay,
            "gameplay_hold_delay": form.gameplay_hold_delay,
        })
        return jsonify({"success": True}), 200
    ```

    Route 2: POST /automation/pause_resume — toggle automation active state
    ```python
    @app.route("/automation/pause_resume", methods=["POST"])
    def automation_pause_resume():
        payload = request.get_json(silent=True) or request.form.to_dict()
        action = (payload.get("action") or "").strip().lower()
        if action not in ("pause", "resume"):
            return jsonify({"success": False, "error": "Invalid action. Use 'pause' or 'resume'."}), 200

        auto_controller = getattr(app, "_auto_controller", None)
        if auto_controller is None:
            return jsonify({"success": False, "error": "Auto controller not available"}), 200

        active = action == "resume"
        auto_controller.set_automation_active(active)
        return jsonify({"success": True, "active": active}), 200
    ```

    Route 3: POST /automation/proceed — advance a -1 delay pause
    ```python
    @app.route("/automation/proceed", methods=["POST"])
    def automation_proceed():
        auto_controller = getattr(app, "_auto_controller", None)
        if auto_controller is None:
            return jsonify({"success": False, "error": "Auto controller not available"}), 200

        ok = auto_controller.proceed()
        if ok:
            return jsonify({"success": True}), 200
        return jsonify({"success": False, "error": "No pending transition to proceed"}), 200
    ```

    Route 4: POST /automation/emergency_live — trigger Emergency Live
    ```python
    @app.route("/automation/emergency_live", methods=["POST"])
    def automation_emergency_live():
        auto_controller = getattr(app, "_auto_controller", None)
        if auto_controller is None:
            return jsonify({"success": False, "error": "Auto controller not available"}), 200

        ok = auto_controller.emergency_live()
        if ok:
            return jsonify({"success": True}), 200
        return jsonify({"success": False, "error": "Failed to switch to Live scene"}), 200
    ```
