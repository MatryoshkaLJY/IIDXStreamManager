    The controller tracks the last known state of all 4 cabinets and triggers scene transitions based on aggregate state:

    ```python
    import json
    import logging
    import threading
    from typing import Any, Dict, Optional

    from flask_socketio import SocketIO

    from src.obs.scene_controller import REQUIRED_SCENES, SceneController
    from src.state import RuntimeState, load_runtime_state, save_runtime_state

    logger = logging.getLogger(__name__)

    DEFAULT_MACHINES = ["IIDX#1", "IIDX#2", "IIDX#3", "IIDX#4"]

    class AutoTransitionController:
        """Background worker that triggers OBS scene transitions based on aggregate cabinet state."""

        def __init__(
            self,
            socketio: SocketIO,
            scene_controller: SceneController,
            interval: float = 1.0,
        ) -> None:
            self.socketio = socketio
            self.scene_controller = scene_controller
            self.interval = interval
            self._stop = threading.Event()
            self._thread: threading.Thread | None = None
            self._machine_states: Dict[str, str] = {}
            self._gameplay_entered_at: Optional[float] = None
            self._pending_transition: Optional[Dict[str, Any]] = None
            self._paused_reason: str = ""

        def start(self) -> None:
            if self._thread is not None and self._thread.is_alive():
                return
            self._stop.clear()
            self._thread = threading.Thread(target=self._run, daemon=True)
            self._thread.start()

        def stop(self) -> None:
            self._stop.set()
            if self._thread is not None:
                self._thread.join(timeout=self.interval + 1)

        def _run(self) -> None:
            while not self._stop.is_set():
                self._tick()
                self._stop.wait(self.interval)

        def _tick(self) -> None:
            runtime_state = load_runtime_state()

            # Check prerequisites (D-16)
            prereq_ok, reason = self._check_prerequisites(runtime_state)
            if not prereq_ok:
                if runtime_state.automation_state != "emergency":
                    self._set_automation_state(runtime_state, "paused", reason)
                return

            # If emergency, stay paused
            if runtime_state.automation_state == "emergency":
                return

            # If operator paused automation (not -1 delay), stay paused
            if not runtime_state.automation_active:
                self._set_automation_state(runtime_state, "paused", "Operator paused automation")
                return

            # If there's a pending -1 delay transition, stay paused
            if self._pending_transition is not None:
                self._set_automation_state(
                    runtime_state, "paused",
                    f"Paused: waiting for operator Proceed ({self._pending_transition['name']})"
                )
                return

            # Determine current aggregate state
            all_play = all(self._machine_states.get(m) == "play" for m in DEFAULT_MACHINES)
            all_live_or_blank = all(
                self._machine_states.get(m) in ("live", "blank", "")
                for m in DEFAULT_MACHINES
            )
            any_assigned = any(
                runtime_state.cabinet_assignments.get(m, "Unassigned") != "Unassigned"
                for m in DEFAULT_MACHINES
            )

            if not any_assigned:
                self._set_automation_state(runtime_state, "paused", "No cabinets assigned")
                return

            # D-01: Live -> Gameplay when all cabinets enter play
            if all_play:
                self._set_automation_state(runtime_state, "active", "")
                # Determine gameplay scene based on mode
                gameplay_scene = self._resolve_gameplay_scene(runtime_state)
                if gameplay_scene:
                    ok, error = self.scene_controller.switch_to(gameplay_scene)
                    if ok:
                        self._gameplay_entered_at = time.time()
                        logger.info("Auto-transition: Live -> %s", gameplay_scene)
                        self.socketio.emit("auto_transition", {
                            "from": "现场摄像",
                            "to": gameplay_scene,
                            "reason": "All cabinets in play",
                        })
                return

            # D-02: Gameplay -> Live when all cabinets return to live/blank
            if all_live_or_blank and self._gameplay_entered_at is not None:
                hold_delay = runtime_state.gameplay_hold_delay
                elapsed = time.time() - self._gameplay_entered_at

                if hold_delay == -1.0:
                    # D-07: -1 means pause for manual proceed
                    self._pending_transition = {
                        "name": "Gameplay -> Live",
                        "target_scene": "现场摄像",
                        "estimated_delay": 5.0,
                    }
                    self._set_automation_state(
                        runtime_state, "paused",
                        "Paused: waiting for operator Proceed (Gameplay -> Live)"
                    )
                    self.socketio.emit("proceed_required", self._pending_transition)
                    return

                if elapsed >= hold_delay:
                    ok, error = self.scene_controller.switch_to("现场摄像")
                    if ok:
                        self._gameplay_entered_at = None
                        logger.info("Auto-transition: Gameplay -> Live")
                        self.socketio.emit("auto_transition", {
                            "from": "Gameplay",
                            "to": "现场摄像",
                            "reason": "All cabinets returned to live/blank",
                        })
                else:
                    self._set_automation_state(
                        runtime_state, "active",
                        f"Holding gameplay for {hold_delay - elapsed:.1f}s"
                    )
                return

            # Default: active with no pending transition
            self._set_automation_state(runtime_state, "active", "")

        def _check_prerequisites(self, runtime_state: RuntimeState) -> tuple[bool, str]:
            if not runtime_state.monitoring_active:
                return False, "Monitoring not active"
            if not runtime_state.obs_connected:
                return False, "OBS not connected"
            if not self.scene_controller.scenes_valid:
                return False, "OBS scenes not valid"
            return True, ""

        def _set_automation_state(self, runtime_state: RuntimeState, state: str, reason: str) -> None:
            if runtime_state.automation_state != state or self._paused_reason != reason:
                runtime_state.automation_state = state
                self._paused_reason = reason
                save_runtime_state(runtime_state)
                self.socketio.emit("automation_status", {
                    "state": state,
                    "reason": reason,
                })

        def _resolve_gameplay_scene(self, runtime_state: RuntimeState) -> Optional[str]:
            mode = runtime_state.mode
            if mode == "team":
                # Determine SP vs DP from current round in schedule
                # For now, default to SP团队赛
                return REQUIRED_SCENES["sp_team"]
            elif mode == "individual":
                return REQUIRED_SCENES["individual"]
            return None

        def on_cabinet_update(self, data: Dict[str, Any]) -> None:
            """Called externally when a cabinet_update event is received."""
            machine_id = data.get("machine_id")
            state = data.get("state", "")
            if machine_id:
                self._machine_states[machine_id] = state

        def proceed(self) -> bool:
            """Operator clicked Proceed on a -1 delay pause."""
            if self._pending_transition is None:
                return False
            target = self._pending_transition["target_scene"]
            ok, error = self.scene_controller.switch_to(target)
            if ok:
                self.socketio.emit("auto_transition", {
                    "from": "Paused",
                    "to": target,
                    "reason": "Operator proceeded",
                })
                if target == "现场摄像":
                    self._gameplay_entered_at = None
            self._pending_transition = None
            return ok

        def emergency_live(self) -> bool:
            """Operator triggered Emergency Live."""
            runtime_state = load_runtime_state()
            ok, error = self.scene_controller.switch_to("现场摄像")
            if ok:
                runtime_state.automation_state = "emergency"
                save_runtime_state(runtime_state)
                self._pending_transition = None
                self._gameplay_entered_at = None
                self.socketio.emit("automation_status", {
                    "state": "emergency",
                    "reason": "Emergency: operator triggered Emergency Live",
                })
                self.socketio.emit("auto_transition", {
                    "from": "Any",
                    "to": "现场摄像",
                    "reason": "Emergency Live",
                })
            return ok

        def set_automation_active(self, active: bool) -> None:
            """Operator toggled pause/resume."""
            runtime_state = load_runtime_state()
            runtime_state.automation_active = active
            if active and runtime_state.automation_state == "emergency":
                runtime_state.automation_state = "paused"
            save_runtime_state(runtime_state)
            self.socketio.emit("automation_status", {
                "state": runtime_state.automation_state,
                "reason": "" if active else "Operator paused automation",
            })
    ```
