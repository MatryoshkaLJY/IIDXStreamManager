    1. Add to static/js/operator.js (append to the IIFE, after existing code):

    ```javascript
    // Automation UI
    const automationBanner = document.getElementById('automation-banner');
    const automationBannerText = document.getElementById('automation-banner-text');
    const automationStateLabel = document.getElementById('automation-state-label');
    const automationReason = document.getElementById('automation-reason');
    const proceedBar = document.getElementById('proceed-bar');
    const proceedText = document.getElementById('proceed-text');
    const proceedBtn = document.getElementById('proceed-btn');
    const emergencyLiveBtn = document.getElementById('emergency-live-btn');
    const automationPauseBtn = document.getElementById('automation-pause-btn');
    const automationActionInput = document.getElementById('automation-action');

    let automationState = window.initialAutomationState || 'paused';
    let automationActive = window.initialAutomationActive !== false;

    function setAutomationUI(state, reason) {
      automationState = state;
      if (!automationBanner || !automationBannerText || !automationStateLabel) return;
      automationBanner.style.display = 'block';
      automationStateLabel.textContent = 'Automation: ' + (state.charAt(0).toUpperCase() + state.slice(1));
      if (state === 'active') {
        automationBannerText.className = 'status-ok';
      } else if (state === 'emergency') {
        automationBannerText.className = 'status-error';
      } else {
        automationBannerText.className = 'status-warning';
      }
      if (automationReason) {
        automationReason.textContent = reason ? ' — ' + reason : '';
      }
    }

    function setPauseButtonUI(active) {
      automationActive = active;
      if (!automationPauseBtn || !automationActionInput) return;
      automationPauseBtn.textContent = active ? 'Pause Automation' : 'Resume Automation';
      automationPauseBtn.className = active ? 'status-ok' : 'status-warning';
      automationActionInput.value = active ? 'pause' : 'resume';
    }

    // Initial state
    setAutomationUI(automationState, '');

    socket.on('automation_status', (data) => {
      setAutomationUI(data.state || 'paused', data.reason || '');
      setPauseButtonUI(data.state === 'active');
    });

    socket.on('proceed_required', (data) => {
      if (!proceedBar || !proceedText) return;
      proceedBar.style.display = 'block';
      const name = data.name || 'Pending transition';
      const delay = typeof data.estimated_delay === 'number' ? data.estimated_delay : 5.0;
      proceedText.textContent = `${name} — Click Proceed to advance (would be ${delay}s)`;
    });

    socket.on('auto_transition', () => {
      if (proceedBar) proceedBar.style.display = 'none';
    });

    if (proceedBtn) {
      proceedBtn.addEventListener('click', () => {
        fetch('/automation/proceed', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({}),
        })
          .then((resp) => resp.json())
          .then((data) => {
            if (data && data.success) {
              if (proceedBar) proceedBar.style.display = 'none';
            }
          })
          .catch((err) => console.error('Proceed error', err));
      });
    }

    if (emergencyLiveBtn) {
      emergencyLiveBtn.addEventListener('click', () => {
        if (!confirm('Trigger Emergency Live? This will cut to the Live scene immediately.')) return;
        fetch('/automation/emergency_live', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({}),
        })
          .then((resp) => resp.json())
          .then((data) => {
            if (data && data.success) {
              setAutomationUI('emergency', 'Emergency: operator triggered Emergency Live');
            }
          })
          .catch((err) => console.error('Emergency Live error', err));
      });
    }

    // Automation delays form (on automation page)
    const delaysForm = document.getElementById('automation-delays-form');
    if (delaysForm) {
      delaysForm.addEventListener('submit', (event) => {
        event.preventDefault();
        const formData = new FormData(delaysForm);
        const payload = {
          return_to_live_delay: parseFloat(formData.get('return_to_live_delay') || '5'),
          gameplay_hold_delay: parseFloat(formData.get('gameplay_hold_delay') || '3'),
        };
        fetch(delaysForm.action, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload),
        })
          .then((resp) => resp.json())
          .then((data) => {
            if (data && data.success) {
              console.log('Delays saved');
            }
          })
          .catch((err) => console.error('Delays save error', err));
      });
    }

    // Pause/resume form (on automation and status pages)
    const pauseForm = document.getElementById('automation-pause-form');
    if (pauseForm) {
      pauseForm.addEventListener('submit', (event) => {
        event.preventDefault();
        const formData = new FormData(pauseForm);
        const payload = { action: formData.get('action') || 'pause' };
        fetch(pauseForm.action, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload),
        })
          .then((resp) => resp.json())
          .then((data) => {
            if (data && typeof data.active === 'boolean') {
              setPauseButtonUI(data.active);
            }
          })
          .catch((err) => console.error('Pause/resume error', err));
      });
    }

    socket.on('automation_delays_updated', (data) => {
      const rtlInput = document.getElementById('return-to-live-delay');
      const ghInput = document.getElementById('gameplay-hold-delay');
      if (rtlInput && typeof data.return_to_live_delay === 'number') {
        rtlInput.value = data.return_to_live_delay;
      }
      if (ghInput && typeof data.gameplay_hold_delay === 'number') {
        ghInput.value = data.gameplay_hold_delay;
      }
    });
    ```
